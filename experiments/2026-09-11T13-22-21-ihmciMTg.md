def build_tokenizer(train_bytes):
    """Train a byte-level BPE tokenizer on a distributed corpus sample.

    Improvements:
    - Uses a larger, distributed sample for better merge statistics.
    - Uses deterministic stratified-random sampling instead of always
      taking the midpoint of each region.
    - Prevents merges across artificial sampled-chunk boundaries.
    - Handles overlapping self-pairs correctly.
    - Preserves deterministic BPE tie-breaking.
    """
    import numpy as np

    NUM_MERGES = 1536

    # 4 MB gives much more reliable statistics for 1536 merges.
    # Drop back to 1_000_000 if training speed is more important.
    SAMPLE_SIZE = 4_000_000

    # More chunks = better coverage across a large corpus.
    NUM_CHUNKS = 128

    RANDOM_SEED = 12345

    corpus_size = len(train_bytes)

    if corpus_size < 2:
        return []

    # ------------------------------------------------------------
    # Build representative training sample
    # ------------------------------------------------------------

    if corpus_size <= SAMPLE_SIZE:
        arr = np.frombuffer(
            train_bytes,
            dtype=np.uint8,
        ).astype(np.int32)

        segments = np.zeros(arr.size, dtype=np.uint16)

    else:
        rng = np.random.default_rng(RANDOM_SEED)

        # Make the sample size exactly divisible by NUM_CHUNKS.
        chunk_size = SAMPLE_SIZE // NUM_CHUNKS

        bounds = np.linspace(
            0,
            corpus_size,
            NUM_CHUNKS + 1,
            dtype=np.int64,
        )

        chunks = []
        segment_chunks = []

        for segment_id in range(NUM_CHUNKS):
            lo = int(bounds[segment_id])
            hi = int(bounds[segment_id + 1])

            available = hi - lo

            if available <= chunk_size:
                start = lo
                size = available
            else:
                # Random location inside this stratum.
                #
                # Fixed RNG seed keeps tokenizer training reproducible,
                # while avoiding systematic midpoint sampling.
                start = int(
                    rng.integers(
                        lo,
                        hi - chunk_size + 1,
                    )
                )
                size = chunk_size

            if size == 0:
                continue

            chunk = np.frombuffer(
                train_bytes,
                dtype=np.uint8,
                count=size,
                offset=start,
            ).astype(np.int32)

            chunks.append(chunk)

            segment_chunks.append(
                np.full(
                    size,
                    segment_id,
                    dtype=np.uint16,
                )
            )

        arr = np.concatenate(chunks)
        segments = np.concatenate(segment_chunks)

    merges = []

    # ------------------------------------------------------------
    # Greedy BPE training
    # ------------------------------------------------------------

    for merge_idx in range(NUM_MERGES):
        n = arr.size

        if n < 2:
            break

        vocab_size = 256 + merge_idx

        # Adjacent tokens may only form a pair if they originated
        # from the same sampled chunk.
        same_segment = segments[:-1] == segments[1:]

        if not same_segment.any():
            break

        # Encode (a, b) into one integer.
        pair_ids = (
            arr[:-1].astype(np.int64) * vocab_size
            + arr[1:]
        )

        # Ignore artificial boundaries.
        #
        # Rather than introducing vocab_size**2 as a sentinel and
        # forcing bincount() to allocate space for it, map boundaries
        # to zero and subtract them afterward.
        num_boundaries = int((~same_segment).sum())

        pair_ids[~same_segment] = 0

        counts = np.bincount(pair_ids)

        if num_boundaries:
            counts[0] -= num_boundaries

        best_pair_id = int(counts.argmax())
        best_pair_count = int(counts[best_pair_id])

        if best_pair_count < 2:
            break

        a, b = divmod(best_pair_id, vocab_size)

        # --------------------------------------------------------
        # Locate every occurrence of the winning pair
        # --------------------------------------------------------

        positions = np.flatnonzero(
            same_segment
            & (arr[:-1] == a)
            & (arr[1:] == b)
        )

        # Self-pairs can overlap.
        #
        # Example:
        #     a a a a
        #
        # Pair locations are:
        #     0, 1, 2
        #
        # but valid simultaneous merges are only:
        #     0, 2
        if a == b and positions.size > 1:
            new_run = np.r_[
                True,
                np.diff(positions) > 1,
            ]

            run_starts = np.maximum.accumulate(
                np.where(
                    new_run,
                    np.arange(positions.size),
                    0,
                )
            )

            offsets = (
                np.arange(positions.size)
                - run_starts
            )

            positions = positions[offsets % 2 == 0]

        if positions.size == 0:
            break

        # Record merge before modifying the stream.
        merges.append((a, b))

        new_token = 256 + merge_idx

        # Replace left side of every occurrence.
        arr[positions] = new_token

        # Remove right side.
        keep = np.ones(arr.size, dtype=bool)
        keep[positions + 1] = False

        arr = arr[keep]
        segments = segments[keep]

    return merges
