def build_tokenizer(train_bytes):
    """
    Improved BPE tokenizer: larger sample (4 MB) and more merges (up to 1792)
    to achieve better compression within the vocab cap of 2048.
    """
    import numpy as np

    # Max merges allowed: 2048 vocab cap - 256 base tokens = 1792
    MAX_MERGES = 1792
    SAMPLE_SIZE = 4_000_000          # 4 MB sample for better coverage
    NUM_CHUNKS = 256                 # More chunks → distributed sampling

    corpus_size = len(train_bytes)

    if corpus_size < 2:
        return []

    # Sample from the corpus, either whole or distributed chunks
    if corpus_size <= SAMPLE_SIZE:
        arr = np.frombuffer(train_bytes, dtype=np.uint8).astype(np.int32)
        segments = np.zeros(len(arr), dtype=np.uint8)
    else:
        chunk_size = SAMPLE_SIZE // NUM_CHUNKS
        bounds = np.linspace(0, corpus_size, NUM_CHUNKS + 1, dtype=np.int64)

        chunks = []
        segment_chunks = []

        for segment_id, (lo, hi) in enumerate(zip(bounds[:-1], bounds[1:])):
            # Take a contiguous block from the middle of each chunk
            start = int(lo + (hi - lo - chunk_size) // 2)
            chunk = np.frombuffer(train_bytes, dtype=np.uint8, count=chunk_size, offset=start).astype(np.int32)
            chunks.append(chunk)
            segment_chunks.append(np.full(chunk_size, segment_id, dtype=np.uint8))

        arr = np.concatenate(chunks)
        segments = np.concatenate(segment_chunks)

    merges = []

    for i in range(MAX_MERGES):
        if len(arr) < 2:
            break

        vocab_size = 256 + i
        same_segment = (segments[:-1] == segments[1:])

        if not np.any(same_segment):
            break

        # Pair IDs, cross‑segment pairs are ignored
        pair_ids = arr[:-1] * vocab_size + arr[1:]
        boundary_id = vocab_size * vocab_size
        pair_ids[~same_segment] = boundary_id

        counts = np.bincount(pair_ids, minlength=boundary_id + 1)
        counts[boundary_id] = 0

        best_pair_id = int(counts.argmax())
        best_pair_count = int(counts[best_pair_id])

        if best_pair_count < 2:
            break

        a, b = divmod(best_pair_id, vocab_size)

        # Locate all occurrences of the best pair
        positions = np.flatnonzero(same_segment & (arr[:-1] == a) & (arr[1:] == b))

        # For identical tokens, merge only non‑overlapping pairs
        if a == b and len(positions) > 1:
            new_run = np.r_[True, np.diff(positions) > 1]
            run_starts = np.maximum.accumulate(np.where(new_run, np.arange(len(positions)), 0))
            offsets = np.arange(len(positions)) - run_starts
            positions = positions[offsets % 2 == 0]

        merges.append((a, b))

        # Replace first element with new token id and drop the second
        arr[positions] = 256 + i
        keep = np.ones(len(arr), dtype=bool)
        keep[positions + 1] = False
        arr = arr[keep]
        segments = segments[keep]

    return merges
