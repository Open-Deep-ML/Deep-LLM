class Attention(nn.Module):
    """Multi-head causal self-attention with RoPE + MidNorm."""

    def __init__(self, cfg):
        super().__init__()
        assert cfg.n_embd % cfg.n_head == 0

        self.n_head = cfg.n_head
        self.head_dim = cfg.n_embd // cfg.n_head

        self.qkv = nn.Linear(
            cfg.n_embd,
            3 * cfg.n_embd,
            bias=False,
        )

        self.proj = nn.Linear(
            cfg.n_embd,
            cfg.n_embd,
            bias=False,
        )

        # MidNorm: normalize attention branch before residual addition
        self.mid_norm = nn.RMSNorm(cfg.n_embd)

        self.attn_dropout = cfg.dropout

        inv_freq = 1.0 / (
            10000 ** (
                torch.arange(0, self.head_dim, 2).float()
                / self.head_dim
            )
        )

        t = torch.arange(cfg.block_size).float()
        freqs = torch.outer(t, inv_freq)

        self.register_buffer(
            "cos", freqs.cos(), persistent=False
        )
        self.register_buffer(
            "sin", freqs.sin(), persistent=False
        )

    def _apply_rope(self, x, T):
        # x: (B, n_head, T, head_dim)

        cos = self.cos[:T].unsqueeze(0).unsqueeze(0)
        sin = self.sin[:T].unsqueeze(0).unsqueeze(0)

        x1 = x[..., 0::2]
        x2 = x[..., 1::2]

        rot_x1 = x1 * cos - x2 * sin
        rot_x2 = x1 * sin + x2 * cos

        return torch.stack(
            [rot_x1, rot_x2],
            dim=-1,
        ).flatten(-2)

    def forward(self, x):
        B, T, C = x.shape

        q, k, v = self.qkv(x).split(C, dim=2)

        q = q.view(
            B, T, self.n_head, self.head_dim
        ).transpose(1, 2)

        k = k.view(
            B, T, self.n_head, self.head_dim
        ).transpose(1, 2)

        v = v.view(
            B, T, self.n_head, self.head_dim
        ).transpose(1, 2)

        q = self._apply_rope(q, T)
        k = self._apply_rope(k, T)

        y = F.scaled_dot_product_attention(
            q,
            k,
            v,
            is_causal=True,
            dropout_p=(
                self.attn_dropout
                if self.training
                else 0.0
            ),
            scale=8.0 / self.head_dim,
        )

        y = (
            y.transpose(1, 2)
            .contiguous()
            .view(B, T, C)
        )

        # Attention output projection
        y = self.proj(y)

        # MidNorm BEFORE residual addition
        return self.mid_norm(y)

class Block(nn.Module):
    """Pre-norm transformer block (vanilla nanoGPT wiring)."""

    def __init__(self, cfg):
        super().__init__()
        self.ln1 = Norm(cfg.n_embd)
        self.attn = Attention(cfg)
        self.ln2 = Norm(cfg.n_embd)
        self.ffn = FFN(cfg)

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        return x + self.ffn(self.ln2(x))


class GPT(nn.Module):
    """Embeddings -> N blocks -> final norm -> tied lm_head."""

    def __init__(self, cfg):
        super().__init__()
        self.embeddings = Embeddings(cfg)

        self.blocks = nn.ModuleList(
            [Block(cfg) for _ in range(cfg.n_layer)]
        )

        self.norm_f = Norm(cfg.n_embd)

        self.lm_head = nn.Linear(
            cfg.n_embd,
            cfg.vocab_size,
            bias=False,
        )

        # μP-style initialization
        self.apply(self._init_weights)

        # Optional depth multiplier.
        #
        # This is compatible with μP as long as n_layer is not the
        # width dimension you are transferring across. It is just a
        # width-independent residual multiplier.
        residual_mult = 1.0 / math.sqrt(2.0 * cfg.n_layer)

        for block in self.blocks:
            # Residual-producing projections.
            #
            # μP hidden matrices use fan-in scaling:
            #     Var(W) ~ 1 / fan_in
            #
            # We retain the GPT-style depth correction as an
            # additional constant multiplier.
            self._init_linear(
                block.attn.proj,
                mult=residual_mult,
            )

            self._init_linear(
                block.ffn.down,
                mult=residual_mult,
            )

        # Tied input/output embedding.
        self.lm_head.weight = self.embeddings.tok.weight

    @staticmethod
    def _init_linear(module, mult=1.0):
        fan_in = module.weight.shape[1]

        nn.init.normal_(
            module.weight,
            mean=0.0,
            std=mult / math.sqrt(fan_in),
        )

        if module.bias is not None:
            nn.init.zeros_(module.bias)

    @staticmethod
    def _init_weights(module):
        if isinstance(module, nn.Linear):
            # μP Transformer hidden weights:
            # Var(W) ∝ 1 / fan_in
            fan_in = module.weight.shape[1]

            nn.init.normal_(
                module.weight,
                mean=0.0,
                std=1.0 / math.sqrt(fan_in),
            )

            if module.bias is not None:
                nn.init.zeros_(module.bias)

        elif isinstance(module, nn.Embedding):
            # Vocabulary size does not scale with model width.
            #
            # μP therefore treats word embeddings as input weights
            # whose initialization variance stays O(1) as d_model
            # increases.
            nn.init.normal_(
                module.weight,
                mean=0.0,
                std=1.0,
            )

    def forward(self, idx):
        x = self.embeddings(idx)

        for block in self.blocks:
            x = block(x)

        return self.lm_head(self.norm_f(x))


def build_model(cfg):
    return GPT(cfg)
