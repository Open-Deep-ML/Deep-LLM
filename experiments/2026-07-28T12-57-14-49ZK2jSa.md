class Block(nn.Module):
    """Parallel attention/SwiGLU block with fused input and output projections."""

    def __init__(self, cfg):
        super().__init__()
        assert cfg.n_embd % cfg.n_head == 0

        self.n_head = cfg.n_head
        self.head_dim = cfg.n_embd // cfg.n_head
        assert self.head_dim % 2 == 0

        hidden_dim = int((8 / 3) * cfg.n_embd)
        hidden_dim = 64 * ((hidden_dim + 63) // 64)

        self.hidden_dim = hidden_dim
        self.ln = Norm(cfg.n_embd)

        # Fused projection:
        #
        # Q, K, V:        3 * n_embd
        # gate, up:       2 * hidden_dim
        self.qkv_gate_up = nn.Linear(
            cfg.n_embd,
            3 * cfg.n_embd + 2 * hidden_dim,
            bias=False,
        )

        # Fused output projection:
        #
        # attention:      n_embd
        # SwiGLU:         hidden_dim
        self.proj_down = nn.Linear(
            cfg.n_embd + hidden_dim,
            cfg.n_embd,
            bias=False,
        )

        self.attn_dropout = cfg.dropout
        self.drop = nn.Dropout(cfg.dropout)

        # Precompute RoPE frequencies.
        inv_freq = 1.0 / (
            10000 ** (
                torch.arange(
                    0,
                    self.head_dim,
                    2,
                ).float() / self.head_dim
            )
        )

        t = torch.arange(cfg.block_size).float()
        freqs = torch.outer(t, inv_freq)

        self.register_buffer(
            "cos",
            freqs.cos(),
            persistent=False,
        )

        self.register_buffer(
            "sin",
            freqs.sin(),
            persistent=False,
        )

    def _apply_rope(self, x, T):
        # x: (B, n_head, T, head_dim)
        cos = self.cos[:T].unsqueeze(0).unsqueeze(0)
        sin = self.sin[:T].unsqueeze(0).unsqueeze(0)

        x1 = x[..., 0::2]
        x2 = x[..., 1::2]

        rot_x1 = x1 * cos - x2 * sin
        rot_x2 = x1 * sin + x2 * cos

        return torch.stack(
            [rot_x1, rot_x2],
            dim=-1,
        ).flatten(-2)

    def forward(self, x):
        B, T, C = x.shape

        z = self.ln(x)

        q, k, v, gate, up = self.qkv_gate_up(z).split(
            [
                C,
                C,
                C,
                self.hidden_dim,
                self.hidden_dim,
            ],
            dim=2,
        )

        q = q.view(
            B,
            T,
            self.n_head,
            self.head_dim,
        ).transpose(1, 2)

        k = k.view(
            B,
            T,
            self.n_head,
            self.head_dim,
        ).transpose(1, 2)

        v = v.view(
            B,
            T,
            self.n_head,
            self.head_dim,
        ).transpose(1, 2)

        q = self._apply_rope(q, T)
        k = self._apply_rope(k, T)

        attn = F.scaled_dot_product_attention(
            q,
            k,
            v,
            is_causal=True,
            dropout_p=(
                self.attn_dropout
                if self.training
                else 0.0
            ),
        )

        attn = (
            attn
            .transpose(1, 2)
            .contiguous()
            .view(B, T, C)
        )

        ffn = F.silu(gate) * up

        # Equivalent to:
        #
        # attention_output @ W_o
        #     +
        # swiglu_output @ W_down
        #
        # but implemented as one output GEMM.
        update = self.proj_down(
            torch.cat(
                [attn, ffn],
                dim=2,
            )
        )

        return x + self.drop(update)


class GPT(nn.Module):
    """Embeddings -> N fused parallel blocks -> final norm -> tied lm_head."""

    def __init__(self, cfg):
        super().__init__()

        self.embeddings = Embeddings(cfg)

        self.blocks = nn.ModuleList(
            [Block(cfg) for _ in range(cfg.n_layer)]
        )

        self.norm_f = Norm(cfg.n_embd)

        self.lm_head = nn.Linear(
            cfg.n_embd,
            cfg.vocab_size,
            bias=False,
        )

        self.apply(self._init_weights)

        residual_std = 0.02 / math.sqrt(
            2.0 * cfg.n_layer
        )

        for block in self.blocks:
            # proj_down contains both the attention output
            # projection and the FFN down projection.
            nn.init.normal_(
                block.proj_down.weight,
                mean=0.0,
                std=residual_std,
            )

        self.lm_head.weight = self.embeddings.tok.weight

    @staticmethod
    def _init_weights(module):
        if isinstance(module, nn.Linear):
            nn.init.normal_(
                module.weight,
                mean=0.0,
                std=0.02,
            )

            if module.bias is not None:
                nn.init.zeros_(module.bias)

        elif isinstance(module, nn.Embedding):
            nn.init.normal_(
                module.weight,
                mean=0.0,
                std=0.02,
            )

    def forward(self, idx):
        x = self.embeddings(idx)

        for block in self.blocks:
            x = block(x)

        return self.lm_head(
            self.norm_f(x)
        )


def build_model(cfg):
    """Build the fused parallel transformer architecture."""
    return GPT(cfg)
