def _ns5(g, steps=5):
    x = g.bfloat16() if g.is_cuda else g.float()
    transposed = x.size(0) > x.size(1)

    if transposed:
        x = x.T

    x = x / (x.norm() + 1e-7)
    a, b, c = 3.4445, -4.7750, 2.0315

    for _ in range(steps):
        xx = x @ x.T
        x = a * x + (b * xx + c * (xx @ xx)) @ x

    if transposed:
        x = x.T

    return x.to(g.dtype)


def _muonize(update):
    # Orthogonalize fused Q, K and V independently.
    if update.size(0) == 3 * update.size(1):
        chunks = update.chunk(3, dim=0)
    else:
        chunks = (update,)

    results = []

    for chunk in chunks:
        result = _ns5(chunk)
        scale = max(
            1.0,
            chunk.size(0) / chunk.size(1),
        ) ** 0.5
        results.append(result * scale)

    if len(results) == 1:
        return results[0]

    return torch.cat(results, dim=0)


class HybridMuonAdamW(torch.optim.Optimizer):
    def __init__(self, groups):
        super().__init__(groups, {})

    @torch.no_grad()
    def step(self, closure=None):
        loss = None

        if closure is not None:
            with torch.enable_grad():
                loss = closure()

        for group in self.param_groups:
            # The harness writes the scheduled Adam LR to every group.
            lr = group["lr"] * group["lr_mult"]
            wd = group["weight_decay"]

            if group["use_muon"]:
                momentum = group["momentum"]

                for p in group["params"]:
                    if p.grad is None:
                        continue

                    state = self.state[p]

                    if "momentum_buffer" not in state:
                        state["momentum_buffer"] = torch.zeros_like(p)

                    buf = state["momentum_buffer"]
                    buf.lerp_(p.grad, 1.0 - momentum)

                    update = p.grad.lerp(buf, momentum)
                    update = _muonize(update).to(p.dtype)

                    if wd:
                        p.mul_(1.0 - lr * wd)

                    p.add_(update, alpha=-lr)

            else:
                beta1, beta2 = group["betas"]
                eps = group["eps"]

                for p in group["params"]:
                    if p.grad is None:
                        continue

                    state = self.state[p]

                    if not state:
                        state["step"] = 0
                        state["exp_avg"] = torch.zeros_like(p)
                        state["exp_avg_sq"] = torch.zeros_like(p)

                    state["step"] += 1
                    m = state["exp_avg"]
                    v = state["exp_avg_sq"]

                    m.lerp_(p.grad, 1.0 - beta1)
                    v.mul_(beta2).addcmul_(
                        p.grad,
                        p.grad,
                        value=1.0 - beta2,
                    )

                    if wd:
                        p.mul_(1.0 - lr * wd)

                    m_hat = m / (
                        1.0 - beta1 ** state["step"]
                    )
                    v_hat = v / (
                        1.0 - beta2 ** state["step"]
                    )

                    p.addcdiv_(
                        m_hat,
                        v_hat.sqrt().add_(eps),
                        value=-lr,
                    )

        return loss


def configure_optimizer(model, cfg):
    # Hidden transformer matrices use Muon.
    muon = [
        p for p in model.blocks.parameters()
        if p.requires_grad and p.ndim == 2
    ]
    muon_ids = {id(p) for p in muon}

    # Tied token embedding/lm_head remains on AdamW.
    adam_decay = [
        p for p in model.parameters()
        if (
            p.requires_grad
            and p.ndim >= 2
            and id(p) not in muon_ids
        )
    ]

    # Norms and biases remain on AdamW without decay.
    adam_no_decay = [
        p for p in model.parameters()
        if p.requires_grad and p.ndim < 2
    ]

    common_adam = {
        "use_muon": False,
        "lr": cfg.learning_rate,
        "lr_mult": 1.0,
        "betas": (0.9, 0.95),
        "eps": 1e-8,
    }

    return HybridMuonAdamW([
        {
            "params": muon,
            "use_muon": True,
            "lr": cfg.learning_rate,
            "lr_mult": 0.02 / cfg.learning_rate,
            "momentum": 0.95,
            "weight_decay": 0.003,
        },
        {
            "params": adam_decay,
            "weight_decay": 0.05,
            **common_adam,
        },
        {
            "params": adam_no_decay,
            "weight_decay": 0.0,
            **common_adam,
        },
    ])
